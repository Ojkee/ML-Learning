def encode_word(word, prev_tokens):
    def _encode(wrd):
        return [stoi[w] for w in wrd]
    prefix_word = "." * prev_tokens + word
    suffix_word = word + "."
    return list(windowed(_encode(prefix_word), prev_tokens)), _encode(suffix_word)

def encode_words(words, prev_tokens):
    X = []
    Y = []
    for word in words:
        x, y = encode_word(word, prev_tokens)
        X.extend(x)
        Y.extend(y)

    X = torch.tensor(X)
    Y = torch.tensor(Y)
    print(X.shape, Y.shape)
    return X, Y

class FlattenConsecutive:
    def __init__(self, n: int) -> None:
        self._n = n

        self._out: torch.Tensor

    def __call__(self, x: torch.Tensor) -> torch.Tensor:
        B, C, T = x.shape
        y = x.view(B, C // self._n, T * self._n)
        self._out = y.squeeze(dim=1) if y.shape[1] == 1 else y
        return self.out

    @property
    def parameters(self) -> list[torch.Tensor]:
        return []

    @property
    def out(self) -> torch.Tensor:
        return self._out

class Linear:
    def __init__(self,
        fan_in: int,
        fan_out: int,
        bias: bool = True,
    ) -> None:
        self.weights = torch.randn(size=(fan_in, fan_out), generator=GEN) / fan_in ** 0.5
        self.bias = torch.zeros(fan_out) if bias is True else None

        self._out: torch.Tensor

    def __call__(self, x: torch.Tensor) -> torch.Tensor:
        self._out = x @ self.weights
        if self.bias is not None:
            self._out += self.bias
        return self.out

    @property
    def parameters(self) -> list[torch.Tensor]:
        return [self.weights] + ([] if self.bias is None else [self.bias])

    @property
    def out(self) -> torch.Tensor:
        return self._out

class BatchNorm1d:
    def __init__(
        self,
        fan_in: int,
        eps: float = 1e-5,
        momentum: float = 0.001,
        training: bool = True,
    ) -> None:
        self.gamma = torch.ones(fan_in)
        self.beta = torch.zeros(fan_in)

        self.running_mean = torch.zeros(fan_in, requires_grad=False)
        self.running_var = torch.ones(fan_in, requires_grad=False)
        self.momentum = momentum

        self.eps = eps
        self.training = training

        self._out: torch.Tensor

    def __call__(self, x: torch.Tensor) -> torch.Tensor:
        if x.ndim == 3:
            dim = (0, 1)
        elif x.ndim == 2:
            dim = 0
        else:
            raise NotImplementedError()
        x_mean = x.mean(dim=dim, keepdim=True) if self.training else self.running_mean
        x_var = x.var(dim=dim, keepdim=True) if self.training else self.running_var
        x_hat = (x - x_mean) / (x_var + self.eps).sqrt()
        self._out = self.gamma * x_hat + self.beta
        if self.training:
            with torch.no_grad():
                self.running_mean = (1 - self.momentum) * self.running_mean + self.momentum * x_mean
                self.running_var = (1 - self.momentum) * self.running_var + self.momentum * x_var
        return self.out

    @property
    def parameters(self) -> list[torch.Tensor]:
        return [self.gamma, self.beta]

    @property
    def out(self) -> torch.Tensor:
        return self._out

class Sequential:
    def __init__(self, layers: list[Layer]) -> None:
        self._layers = layers

    def __call__(self, x: torch.Tensor) -> torch.Tensor:
        for layer in self._layers:
            x = layer(x)
        self._out = x
        return self.out

    @property
    def parameters(self) -> list[torch.Tensor]:
        return [params for layer in self._layers for params in layer.parameters]

    @property
    def out(self) -> torch.Tensor:
        return self._out

def generate_words(num_words: int, max_len_word: int = 20) -> None:
    for _ in range(num_words):
        block = [0] * PREV_TOKENS
        word = ""
        for _ in range(max_len_word):
            next_pred = model(torch.tensor([block]))
            probs = next_pred.softmax(dim=1)
            next_char_i = int(torch.multinomial(probs, num_samples=1, replacement=True, generator=GEN).item())
            if next_char_i == stoi["."]:
                break
            word += itos[next_char_i]
            block = block[1:] + [next_char_i]
        print(word)
